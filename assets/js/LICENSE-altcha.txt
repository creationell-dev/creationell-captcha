Update procedure (from the plugin root):

  V=3.0.10  # set to the target altcha version
  cd /tmp && rm -rf altcha-vendor && mkdir altcha-vendor && cd altcha-vendor

  # `sfw` is the Socket Firewall wrapper — REQUIRED in this environment
  # to protect against compromised npm packages. Never run plain `npm pack`.
  sfw npm pack altcha@$V

  tar -xzf altcha-$V.tgz

  # Copy the non-locale files into the plugin:
  P=/home/jpk/wp/ALTCHA/creationell-captcha
  cp package/dist/main/altcha.umd.min.cjs   $P/assets/js/altcha.min.js
  cp package/dist/workers/argon2id.js       $P/assets/js/altcha-argon2id.worker.js
  cp package/dist/external/altcha.css       $P/assets/css/altcha.css
  for t in aqua business caramel cupcake cyberpunk lime wireframe; do
      cp package/dist/themes/$t.min.css     $P/assets/css/altcha-themes/$t.css
  done

  for l in de en fr-fr fr-ca es-es es-419 it nl pt-pt pt-br pl cs sk \
           da sv fi nb hu ro el; do
      cp package/dist/i18n/$l.umd.cjs       $P/assets/js/altcha-i18n/$l.js
  done

  cd /tmp && rm -rf altcha-vendor

  # Then refresh this notice (bump the version line at the top).
